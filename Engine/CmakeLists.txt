cmake_minimum_required(VERSION 3.12)

set(PROJECT_NAME Engine)
set(CMAKE_CXX_STANDARD 17)

#--------------PROJECT FILES--------------

set(ENGINE_PRIVATE_INCLUDES
	includes/Application.hpp
	includes/Window.hpp
	includes/VulkanManager.hpp
	includes/SwapChain.hpp
	includes/ValidationLayers.hpp
	includes/VulkanResources.hpp
	includes/VulkanData.hpp
)

set(ENGINE_PRIVATE_SOURCES
	src/Application.cpp
	src/Window.cpp
	src/VulkanManager.cpp
	src/SwapChain.cpp
	src/ValidationLayers.cpp
	src/VulkanResources.cpp
)

set(PROJECT_INCLUDES_SOURCES
	${ENGINE_PRIVATE_INCLUDES}
	${ENGINE_PRIVATE_SOURCES}
)

add_library(${PROJECT_NAME} STATIC ${PROJECT_INCLUDES_SOURCES})
target_include_directories(${PROJECT_NAME} PUBLIC includes)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)

#--------------LIBRARIES--------------

target_link_libraries(${PROJECT_NAME} PUBLIC ConsoleText)

add_subdirectory(external/glfw)
add_subdirectory(external/glm)

target_link_libraries(${PROJECT_NAME} PUBLIC glm 
					     glfw)

include_directories(${PROJECT_NAME} external/stb_image)

#-------------------Vulkan init--------------



find_package(Vulkan REQUIRED)
target_include_directories(${PROJECT_NAME} PUBLIC ${Vulkan_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} PUBLIC Vulkan::Vulkan)
