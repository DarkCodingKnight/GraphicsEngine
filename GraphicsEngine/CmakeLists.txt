cmake_minimum_required(VERSION 3.15)

set(PROJECT_NAME GraphicsEngine)
set(CMAKE_CXX_STANDARD 17)

file(GLOB_RECURSE SOLOUD_AUDIOSOURCES ${SOLOUD_ROOT}/src/audiosource/*.c*)
file(GLOB_RECURSE SOLOUD_CORE ${SOLOUD_ROOT}/src/core/*.c*)
file(GLOB_RECURSE SOLOUD_FILTERS ${SOLOUD_ROOT}/src/filter/*.c*)

#--------------PROJECT FILES--------------

set(ENGINE_PRIVATE_INCLUDES
	includes/GraphicsApplication.hpp
	includes/Window.hpp
	includes/SwapChain.hpp
	includes/ValidationLayers.hpp
	includes/VulkanResources.hpp
	includes/VulkanData.hpp
	includes/Pipeline.hpp
	includes/ResourceManager.hpp
	includes/Renderer.hpp
	includes/GraphicsObject.hpp
	includes/GraphicsObjectPool.hpp
	includes/Mesh.hpp
	includes/Texture.hpp
	includes/SamplerResources.hpp
	includes/DescriptorResources.hpp
	#includes/RenderTargetManager.hpp
	includes/Camera.hpp
	includes/IMGUIwindow.hpp
	includes/IMPLOTwindow.hpp
	includes/Audio.hpp
	includes/Lighting.hpp
)

set(ENGINE_PRIVATE_SOURCES
	src/GraphicsApplication.cpp
	src/Window.cpp
	src/SwapChain.cpp
	src/ValidationLayers.cpp
	src/VulkanResources.cpp
	src/Pipeline.cpp
	src/ResourceManager.cpp
	src/Renderer.cpp
	src/GraphicsObject.cpp
	src/GraphicsObjectPool.cpp
	src/Mesh.cpp
	src/Texture.cpp
	src/SamplerResources.cpp
	src/DescriptorResources.cpp
	#src/RenderTargetManager.cpp
	src/Camera.cpp
	src/IMGUIwindow.cpp
	src/IMPLOTwindow.cpp
	src/Audio.cpp
	src/Lighting.cpp
)

set(PROJECT_INCLUDES_SOURCES
	${ENGINE_PRIVATE_INCLUDES}
	${ENGINE_PRIVATE_SOURCES}
)

add_library(${PROJECT_NAME} STATIC ${PROJECT_INCLUDES_SOURCES})
target_include_directories(${PROJECT_NAME} PUBLIC includes)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)

#--------------LIBRARIES--------------

add_subdirectory(external/glfw)
add_subdirectory(external/glm)
add_subdirectory(external/imgui)
add_subdirectory(external/implot)
add_subdirectory(external/SoLoud)

# Безопасный find_package: MoltenVK — опционально
find_package(Vulkan REQUIRED)

target_link_libraries(${PROJECT_NAME} PUBLIC
    Vulkan::Vulkan
    glfw
    glm
    ConsoleText
    imgui
	implot
    SoLoud
)

# Линкуем MoltenVK только если CMake его реально нашёл
if(TARGET Vulkan::MoltenVK)
    target_link_libraries(${PROJECT_NAME} PUBLIC Vulkan::MoltenVK)
    message(STATUS "Linking against Vulkan::MoltenVK")
else()
    message(STATUS "Vulkan::MoltenVK not found — relying on loader ICD")
endif()

target_include_directories(${PROJECT_NAME} PUBLIC
    ${Vulkan_INCLUDE_DIRS}
    external/stb
    external/tiny_obj_loader
    external/imgui
	external/implot
)
