cmake_minimum_required(VERSION 3.12)

set(PROJECT_NAME Engine)
set(CMAKE_CXX_STANDARD 17)

file(GLOB_RECURSE SOLOUD_AUDIOSOURCES ${SOLOUD_ROOT}/src/audiosource/*.c*)
file(GLOB_RECURSE SOLOUD_CORE ${SOLOUD_ROOT}/src/core/*.c*)
file(GLOB_RECURSE SOLOUD_FILTERS ${SOLOUD_ROOT}/src/filter/*.c*)

#--------------PROJECT FILES--------------

set(ENGINE_PRIVATE_INCLUDES
	includes/GraphicsApplication.hpp
	includes/Window.hpp
	includes/APImanager.hpp
	includes/SwapChain.hpp
	includes/ValidationLayers.hpp
	includes/VulkanResources.hpp
	includes/VulkanData.hpp
	includes/Pipeline.hpp
	includes/ResourceManager.hpp
	includes/Renderer.hpp
	includes/Object.hpp
	includes/ObjectPool.hpp
	includes/Mesh.hpp
)

set(ENGINE_PRIVATE_SOURCES
	src/GraphicsApplication.cpp
	src/Window.cpp
	src/APImanager.cpp
	src/SwapChain.cpp
	src/ValidationLayers.cpp
	src/VulkanResources.cpp
	src/Pipeline.cpp
	src/ResourceManager.cpp
	src/Renderer.cpp
	src/Object.cpp
	src/ObjectPool.cpp
	src/Mesh.cpp
)

set(PROJECT_INCLUDES_SOURCES
	${ENGINE_PRIVATE_INCLUDES}
	${ENGINE_PRIVATE_SOURCES}
)

add_library(${PROJECT_NAME} STATIC ${PROJECT_INCLUDES_SOURCES})
target_include_directories(${PROJECT_NAME} PUBLIC includes)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)

#--------------LIBRARIES--------------

target_link_libraries(${PROJECT_NAME} PUBLIC ConsoleText)

add_subdirectory(external/glfw)
add_subdirectory(external/glm)

find_package(Vulkan REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets)

target_include_directories(${PROJECT_NAME} PUBLIC ${Vulkan_INCLUDE_DIRS} stb_image)
target_link_libraries(${PROJECT_NAME} PUBLIC glm glfw Vulkan::Vulkan stb_image Qt::Widgets)

set(SOLOUD_SOURCES
    ${SOLOUD_ROOT}/external/soloud/src/backend/miniaudio/soloud_miniaudio.cpp
    ${SOLOUD_ROOT}/external/soloud/src/c_api/soloud_c.cpp
    ${SOLOUD_AUDIOSOURCES}
    ${SOLOUD_CORE}
    ${SOLOUD_FILTERS}
)

target_include_directories(${PROJECT_NAME} PUBLIC ${SOLOUD_SOURCES}/includes)
target_compile_definitions(${PROJECT_NAME} PRIVATE WITH_MINIAUDIO)
